import 'package:flutter/material.dart';

class StaticListSearch extends StatefulWidget {
  const StaticListSearch({super.key});

  @override
  State<StaticListSearch> createState() =>
      _StaticListSearchState();
}

class _StaticListSearchState extends State<StaticListSearch> {
  final TextEditingController searchController =
      TextEditingController();

  final List<String> items = [
    'Apple',
    'Banana',
    'Orange',
    'Mango',
    'Pineapple',
    'Grapes',
    'Watermelon',
    'Strawberry',
    'Papaya',
    'Kiwi',
  ];

  String searchText = '';

  List<String> get filteredItems {
    return items
        .where((item) =>
            item.toLowerCase().contains(searchText.toLowerCase()))
        .toList();
  }

  @override
  void dispose() {
    searchController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final results = filteredItems;

    return Scaffold(
      appBar: AppBar(
        title: const Text('Static ListView Search'),
        backgroundColor: Colors.blue,
        foregroundColor: Colors.white,
      ),
      body: Column(
        children: [
          Padding(
            padding: const EdgeInsets.all(12.0),
            child: TextField(
              controller: searchController,
              decoration: InputDecoration(
                hintText: 'Search fruits...',
                prefixIcon: const Icon(Icons.search),
                suffixIcon: searchText.isNotEmpty
                    ? IconButton(
                        icon: const Icon(Icons.clear),
                        onPressed: () {
                          searchController.clear();
                          setState(() {
                            searchText = '';
                          });
                        },
                      )
                    : null,
                border: OutlineInputBorder(
                  borderRadius: BorderRadius.circular(12),
                ),
              ),
              onChanged: (value) {
                setState(() {
                  searchText = value;
                });
              },
            ),
          ),

          Expanded(
            child: results.isEmpty
                ? const Center(
                    child: Text('No items found'),
                  )
                : ListView.builder(
                    itemCount: results.length,
                    itemBuilder: (context, index) {
                      final fruit = results[index];

                      return ListTile(
                        leading: const Icon(
                          Icons.local_florist,
                          color: Colors.green,
                        ),
                        title: Text(fruit),
                        trailing:
                            const Icon(Icons.arrow_forward_ios),
                        onTap: () {
                          ScaffoldMessenger.of(context)
                              .showSnackBar(
                            SnackBar(
                              content: Text('Selected: $fruit'),
                            ),
                          );
                        },
                      );
                    },
                  ),
          ),
        ],
      ),
    );
  }
}
